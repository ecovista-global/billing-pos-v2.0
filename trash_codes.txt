function escapeSqlString(val) {
  if (val === null || val === undefined) return 'NULL';
  if (typeof val === 'number') return val;
  if (typeof val === 'boolean') return val ? 1 : 0;
  return "'" + String(val).replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/\n/g, '\\n').replace(/\r/g, '\\r') + "'";
}

let dataSql = `\n-- =====================================================================\n-- ECOFONE DATA DUMP\n-- =====================================================================\n\n`;

for (const tableName of tables) {
  try {
    const rows = db.prepare(`SELECT * FROM ${tableName}`).all();
    if (rows.length === 0) continue;

    dataSql += `-- ---------------------------------------------------------------------\n`;
    dataSql += `-- Dumping data for table \`${tableName}\` (${rows.length} rows)\n`;
    dataSql += `-- ---------------------------------------------------------------------\n`;
    dataSql += `LOCK TABLES \`${tableName}\` WRITE;\n`;

    const cols = Object.keys(rows[0]);
    const colList = cols.map(c => `\`${c}\``).join(', ');

    // Chunk inserts in batches of 50 for optimal MySQL performance
    const chunkSize = 50;
    for (let i = 0; i < rows.length; i += chunkSize) {
      const chunk = rows.slice(i, i + chunkSize);
      const valuesList = chunk.map(row => {
        const vals = cols.map(c => escapeSqlString(row[c]));
        return `(${vals.join(', ')})`;
      }).join(',\n  ');

      dataSql += `INSERT INTO \`${tableName}\` (${colList}) VALUES\n  ${valuesList};\n`;
    }

    dataSql += `UNLOCK TABLES;\n\n`;
  } catch (err) {
    console.warn(`Could not export data for table ${tableName}:`, err.message);
  }
}

function parseUserPermissions(permissionsStr, role = 'employee') {
  if (role === 'admin') {
    // Admin has full universal permissions for view/edit, but preserve custom flags (like delete)
    let stored = {};
    try {
      stored = typeof permissionsStr === 'string' ? JSON.parse(permissionsStr) : (permissionsStr || {});
    } catch(e) {}
    
    const adminPerms = { ...stored };
    for (const k of Object.keys(DEFAULT_EMPLOYEE_PERMISSIONS)) {
      adminPerms[k] = { ...adminPerms[k], view: true, edit: true };
    }
    return adminPerms;
  }

  if (!permissionsStr) {
    return { ...DEFAULT_EMPLOYEE_PERMISSIONS };
  }

  try {
    const parsed = typeof permissionsStr === 'string' ? JSON.parse(permissionsStr) : permissionsStr;
    const merged = { ...DEFAULT_EMPLOYEE_PERMISSIONS };
    for (const [key, val] of Object.entries(parsed)) {
      if (typeof val === 'object' && val !== null) {
        merged[key] = { ...merged[key], ...val };
      }
    }
    return merged;
  } catch (e) {
    return { ...DEFAULT_EMPLOYEE_PERMISSIONS };
  }
}

db.DEFAULT_EMPLOYEE_PERMISSIONS = DEFAULT_EMPLOYEE_PERMISSIONS;
db.parseUserPermissions = parseUserPermissions;

const fs = require('fs');
const path = require('path');
const trashFile = path.resolve(__dirname, 'trash_code.txt');

function processDir(dir) {
  const files = fs.readdirSync(dir);
  for (const file of files) {
    const fullPath = path.join(dir, file);
    if (fs.statSync(fullPath).isDirectory()) {
      if (file !== 'node_modules' && file !== '.git' && file !== 'dist') {
        processDir(fullPath);
      }
    } else if (fullPath.endsWith('.js') || fullPath.endsWith('.jsx')) {
      let content = fs.readFileSync(fullPath, 'utf8');
      
      // Find console.logs
      const logRegex = /^.*console\.(log|debug|info).*$/gm;
      const logs = content.match(logRegex);
      
      // Find commented blocks that might be dead code (simplistic)
      const commentedBlockRegex = /\/\*[\s\S]*?\*\//g;
      const blocks = content.match(commentedBlockRegex);

      // Find inline commented out JS code (naive heuristic: lines starting with // followed by const, let, var, function, if)
      const inlineCodeRegex = /^\s*\/\/\s*(const|let|var|function|if|return|db\.).*$/gm;
      const inlineCodes = content.match(inlineCodeRegex);

      let found = false;
      let trashOutput = `\n--- From ${fullPath} ---\n`;

      if (logs) {
        found = true;
        trashOutput += "// Logs removed:\n" + logs.join('\n') + "\n";
        content = content.replace(logRegex, '');
      }
      
      if (blocks) {
        found = true;
        trashOutput += "// Commented blocks removed:\n" + blocks.join('\n') + "\n";
        content = content.replace(commentedBlockRegex, '');
      }

      if (inlineCodes) {
        found = true;
        trashOutput += "// Inline commented code removed:\n" + inlineCodes.join('\n') + "\n";
        content = content.replace(inlineCodeRegex, '');
      }

      // Special case for database.js empty inline comments we left earlier
      const deadMigrations = /^\s*\/\/\s*Inline migrations removed.*$/gm;
      const deadMigrationsMatch = content.match(deadMigrations);
      if (deadMigrationsMatch) {
         found = true;
         trashOutput += "// Dead migration comments removed:\n" + deadMigrationsMatch.join('\n') + "\n";
         content = content.replace(deadMigrations, '');
      }

      if (found) {
        fs.appendFileSync(trashFile, trashOutput);
        fs.writeFileSync(fullPath, content);
      }
    }
  }
}

fs.writeFileSync(trashFile, 'DEAD CODE / TRASH / DEBUG LOGS\n=================================\n');

console.log('Cleanup complete. Check trash_code.txt');
